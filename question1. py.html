def ROR(num, n):
print("\nROR Operation")
print("Original:", format(num, '08b'))

for i in range(n):
last = num & 1
num = (num >> 1) | (last << 7)
print("Step", i + 1, ":", format(num, '08b'))

print("Final Result:", format(num, '08b'))


def RCR(num, carry, n):
print("\nRCR Operation")
print("Original:", format(num, '08b'))
print("Initial Carry:", carry)

for i in range(n):
new_carry = num & 1
num = (num >> 1) | (carry << 7)
carry = new_carry

print("Step", i + 1, ":", format(num, '08b'),
"Carry:", carry)

print("Final Result:", format(num, '08b'))
print("Final Carry:", carry)


number = int('11010101', 2)

ROR(number, 3)
RCR(number, 0, 3)
